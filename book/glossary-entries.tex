% Glossary entry definitions. Use \gls{key} in chapters; \glslink{key}{text} inside descriptions.
\newglossaryentry{adjutant}{%
  name={Adjutant},
  description={A resident of \glslink{sindrel}{Sindrel} who holds house patronage but is not a full member of a \glslink{great-house}{Great House}. Adjutants wear gold bracelets bearing their house's mark. See also \glslink{factor}{Factor} and \glslink{silver-guest-bracelet}{silver guest bracelet}},
}

\newglossaryentry{anathemist}{%
  name={Anathemist},
  description={A member of the inter-faith order that holds mortal misuse of magic responsible for the gods' withdrawal and the \glslink{death-of-magic}{death of magic}. Anathemists preach the purging of sorcery from \glslink{arcan}{Arcan}; in recent years they have grown openly hostile to traffic bound for \glslink{sindrel}{Sindrel}},
}

\newglossaryentry{arcan}{%
  name={Arcan},
  description={The world. Magic, its flora and fauna, continental geography, and the presence of a second moon distinguish Arcan from the mundane earth described in pre-magical philosophy},
}

\newglossaryentry{artresh}{%
  name={Artresh},
  description={The northern continent treated at length in Part~II of this guide. Former heartland of the \glslink{magedom-of-houd}{Magedom of Houd}; now a patchwork of successor states, trade routes, and rival courts},
}

\newglossaryentry{death-of-magic}{%
  name={death of magic},
  description={The gradual decline of arcane power across \glslink{arcan}{Arcan}. Also called \glslink{the-fade}{the Fade}. First unmistakable stirrings are dated to $\sim$3850~SC; the process has accelerated within living memory},
}

\newglossaryentry{factor}{%
  name={Factor},
  description={A full member of one of the \glslink{great-house}{Great Houses} of \glslink{sindrel}{Sindrel}. Factors bear an enchanted tattoo of their house sigil, commonly on the hand or forearm. The term is also used loosely for senior house officers. See also \glslink{adjutant}{Adjutant}},
}

\newglossaryentry{the-fade}{%
  name={the Fade},
  description={See \glslink{death-of-magic}{death of magic}},
}

\newglossaryentry{great-house}{%
  name={Great House},
  description={One of the seven institutions that govern \glslink{sindrel}{Sindrel}, each associated with a school of magic and a civic function. Descended, in public account, from the founders of the city},
}

\newglossaryentry{hexine-circle}{%
  name={Hexine Circle},
  description={A sub-council of the \glslink{great-house}{Great Houses} comprising representatives of lesser rank than the house heads. Convened when an issue requires inter-house cooperation but does not warrant the \glslink{octarion-circle}{Octarion Circle}},
}

\newglossaryentry{magedom-of-houd}{%
  name={Magedom of Houd},
  description={The \glslink{thaumocracy}{thaumocratic} empire that once united much of northern \glslink{artresh}{Artresh} under a Mage Lord Council. Shattered after the \glslink{seven-day-war}{Seven Day War} (3958~SC)},
}

\newglossaryentry{mage-road}{%
  name={Mage Road},
  description={The overland route linking \glslink{soulwind}{Soulwind} to \glslink{sindrel}{Sindrel}, and the principal artery of northern trade. Subject to increasing \glslink{anathemist}{Anathemist} predation in the present decade},
}

\newglossaryentry{octarion-circle}{%
  name={Octarion Circle},
  description={The supreme council of \glslink{sindrel}{Sindrel}, comprising the heads of the \glslink{great-house}{Great Houses}. Public tradition holds that an unbroken line of such councils dates to the city's founding (3218~SC)},
}

\newglossaryentry{sc-standard-calendar}{%
  name={SC (Standard Calendar)},
  description={The calendar system widely used across \glslink{arcan}{Arcan}, associated with \glslink{soulwind}{Soulwind}. The present year of this edition is 4218~SC. Also called the \glslink{standard-calendar}{Standard Calendar}},
}

\newglossaryentry{seven-day-war}{%
  name={Seven Day War},
  description={The war of 3958~SC between the \glslink{magedom-of-houd}{Magedom of Houd} and the \glslink{thaumocracy-of-tellerin}{Thaumocracy of Tellerin}, fought in part over scarcity of magical resources. Houdian victory; Tellarin withdrawal and closure of its borders; lasting collapse of Houdian unity},
}

\newglossaryentry{silver-guest-bracelet}{%
  name={silver guest bracelet},
  description={Issued to visitors in \glslink{sindrel}{Sindrel} without house patronage. Permits a stay of thirty-six days; raised, the bracelet displays a map and the days remaining. See also \glslink{adjutant}{Adjutant}},
}

\newglossaryentry{sindrel}{%
  name={Sindrel},
  description={The floating city of the far north, widely regarded as the last place on \glslink{arcan}{Arcan} where the \glslink{death-of-magic}{death of magic} has not taken hold. Centre of a magical trade empire ruled by the \glslink{great-house}{Great Houses}},
}

\newglossaryentry{soulwind}{%
  name={Soulwind},
  description={The largest successor state of old \glslink{magedom-of-houd}{Houd} by area; once a rival centre of magical learning, now a crumbling metropolis in decline since $\sim$3900~SC. Origin of the \glslink{standard-calendar}{Standard Calendar}},
}

\newglossaryentry{standard-calendar}{%
  name={Standard Calendar},
  description={See \glslink{sc-standard-calendar}{SC (Standard Calendar)}},
}

\newglossaryentry{thaumocracy-of-tellerin}{%
  name={Thaumocracy of Tellerin},
  description={The southern \glslink{thaumocracy}{thaumocratic} empire, majority \emph{drow} in common northern memory, famous for necromantic warfare. After defeat in the \glslink{seven-day-war}{Seven Day War}, Tellarin closed its borders; little reliable intelligence has emerged since},
}

\newglossaryentry{thaumocracy}{%
  name={thaumocracy},
  description={Rule by those who wield magical power; government in which arcane strength is the primary basis of legitimacy. Once the ordinary condition of statecraft on \glslink{arcan}{Arcan}; now increasingly a title without substance outside refuges such as \glslink{sindrel}{Sindrel}},
}
