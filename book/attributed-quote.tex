  {%
    \IfValueTF{#2}{%
      \par\vspace{0.35\baselineskip}%
      \hfill\textit{--- #1, #2}%
    }{%
      \IfBlankTF{#1}{}{%
        \par\vspace{0.35\baselineskip}%
        \hfill\textit{--- #1}%
      }%
    }%
    \end{DndReadAloud}%
  }
